cmake_minimum_required(VERSION 3.24)
project(workstation_media_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(workstation-media-probe probe.cpp)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/gstreamer/bin/pkg-config.exe")
  set(PKG_CONFIG_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/gstreamer/bin/pkg-config.exe")
  set(ENV{PKG_CONFIG_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/gstreamer/lib/pkgconfig")
endif()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(GES QUIET IMPORTED_TARGET gst-editing-services-1.0)
  if(GES_FOUND)
    target_link_libraries(workstation-media-probe PRIVATE PkgConfig::GES)
    target_compile_definitions(workstation-media-probe PRIVATE WORKSTATION_HAVE_GES)
    pkg_check_modules(MEDIA REQUIRED IMPORTED_TARGET gstreamer-pbutils-1.0 gstreamer-app-1.0 json-glib-1.0)
    add_executable(workstation-media-worker worker.cpp)
    target_link_libraries(workstation-media-worker PRIVATE PkgConfig::GES PkgConfig::MEDIA)
    target_compile_definitions(workstation-media-worker PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(workstation-media-worker PRIVATE /utf-8 /W4)
    endif()
    if(WIN32)
      pkg_check_modules(D3D11 REQUIRED IMPORTED_TARGET gstreamer-d3d11-1.0)
      add_executable(workstation-gpu-preview gpu-preview.cpp)
      target_link_libraries(workstation-gpu-preview PRIVATE PkgConfig::MEDIA PkgConfig::D3D11 d3d11 dxgi)
      target_compile_definitions(workstation-gpu-preview PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
      target_compile_options(workstation-gpu-preview PRIVATE /utf-8 /W4)
    endif()
  endif()
endif()
